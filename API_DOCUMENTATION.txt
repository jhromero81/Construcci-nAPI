# Proyecto: API de Autenticacion
# Servidor: http://localhost:3001
# Herramienta: Postman
# Fecha: 2026-07-21

# ============================================
# ENDPOINTS DE LA API
# ============================================

# --------------------------------------------
# 1. REGISTRAR USUARIO
# --------------------------------------------
# Metodo:    POST
# URL:       http://localhost:3001/api/register
# Headers:   Content-Type: application/json
#
# Body (JSON):
# {
#     "username": "juan123",
#     "password": "micontrasena"
# }
#
# Respuesta exitosa (201):
# {
#     "mensaje": "Usuario registrado exitosamente",
#     "usuario": {
#         "id": 1,
#         "username": "juan123",
#         "fechaCreacion": "2026-07-21T20:00:00.000Z"
#     }
# }
#
# Error - Campos vacios (400):
# {
#     "error": "Username y password son requeridos"
# }
#
# Error - Usuario existente (400):
# {
#     "error": "El usuario ya existe en el sistema"
# }

# --------------------------------------------
# 2. INICIAR SESION
# --------------------------------------------
# Metodo:    POST
# URL:       http://localhost:3001/api/login
# Headers:   Content-Type: application/json
#
# Body (JSON):
# {
#     "username": "juan123",
#     "password": "micontrasena"
# }
#
# Respuesta exitosa (200):
# {
#     "mensaje": "Autenticacion satisfactoria",
#     "token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
#     "usuario": {
#         "id": 1,
#         "username": "juan123"
#     }
# }
#
# Error - Credenciales incorrectas (401):
# {
#     "error": "Error en la autenticacion: contrasena incorrecta"
# }
#
# Error - Usuario no encontrado (401):
# {
#     "error": "Error en la autenticacion: usuario no encontrado"
# }

# --------------------------------------------
# 3. VERIFICAR ESTADO DEL SERVIDOR
# --------------------------------------------
# Metodo:    GET
# URL:       http://localhost:3001/api/status
# Headers:   No requiere
#
# Respuesta exitosa (200):
# {
#     "mensaje": "API de autenticacion funcionando correctamente",
#     "timestamp": "2026-07-21T20:00:00.000Z",
#     "usuariosRegistrados": 1
# }

# ============================================
# PASOS PARA TESTEAR EN POSTMAN
# ============================================

# PASO 1: Registrar un usuario
#   1. Abrir Postman
#   2. Crear nuevo request
#   3. Seleccionar metodo POST
#   4. Ingresar URL: http://localhost:3001/api/register
#   5. Ir a la pestana "Body"
#   6. Seleccionar "raw" y "JSON"
#   7. Ingresar el JSON con username y password
#   8. Hacer clic en "Send"
#   9. Verificar respuesta 201 Created

# PASO 2: Iniciar sesion
#   1. Crear nuevo request
#   2. Seleccionar metodo POST
#   3. Ingresar URL: http://localhost:3001/api/login
#   4. Ir a la pestana "Body"
#   5. Seleccionar "raw" y "JSON"
#   6. Ingresar el JSON con las credenciales
#   7. Hacer clic en "Send"
#   8. Verificar respuesta 200 OK con token

# PASO 3: Verificar estado
#   1. Crear nuevo request
#   2. Seleccionar metodo GET
#   3. Ingresar URL: http://localhost:3001/api/status
#   4. Hacer clic en "Send"
#   5. Verificar respuesta 200 OK

# ============================================
# CODIGOS DE RESPUESTA HTTP
# ============================================
#
# 200 OK                - Peticion exitosa
# 201 Created           - Recurso creado exitosamente
# 400 Bad Request       - Error en los datos enviados
# 401 Unauthorized      - Credenciales incorrectas
# 404 Not Found         - Recurso no encontrado
# 500 Internal Error    - Error del servidor

# ============================================
# FLUJO DE AUTENTICACION
# ============================================
#
# 1. El usuario se registra con POST /api/register
# 2. El servidor guarda el usuario con contraseña cifrada
# 3. El usuario inicia sesion con POST /api/login
# 4. El servidor verifica credenciales y genera un token JWT
# 5. El frontend guarda el token en localStorage
# 6. El token se usa para autenticar peticiones futuras
